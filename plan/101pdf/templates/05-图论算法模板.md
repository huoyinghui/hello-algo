## 2. 拓扑排序 🔥

```python
from collections import deque, defaultdict

def topological_sort(n, edges):
    """
    Kahn算法 - BFS实现
    n: 节点数量 (0 to n-1)
    edges: [(u, v), ...] 表示 u -> v
    返回: 拓扑序列，如果有环返回[]
    """
    # 构建图和入度数组
    graph = defaultdict(list)
    in_degree = [0] * n
    
    for u, v in edges:
        graph[u].append(v)
        in_degree[v] += 1
    
    # 将所有入度为0的节点入队
    queue = deque([i for i in range(n) if in_degree[i] == 0])
    result = []
    
    while queue:
        node = queue.popleft()
        result.append(node)
        
        # 删除该节点的所有出边
        for neighbor in graph[node]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    
    # 如果结果长度 < n，说明存在环
    return result if len(result) == n else []
```

**时间复杂度：** O(V + E)  
**应用场景：**
- 课程安排
- 任务调度
- 编译顺序

**典型例题：**
- LeetCode 207. 课程表
- LeetCode 210. 课程表 II

## 3. Dijkstra 最短路径 🔥

```python
import heapq

def dijkstra(graph, start):
    """
    Dijkstra算法 - 单源最短路径
    graph: {node: [(neighbor, weight), ...]}
    start: 起点
    返回: {node: distance} 从start到各节点的最短距离
    """
    # 初始化距离
    dist = {node: float('inf') for node in graph}
    dist[start] = 0
    
    # 优先队列 (distance, node)
    pq = [(0, start)]
    visited = set()
    
    while pq:
        d, node = heapq.heappop(pq)
        
        if node in visited:
            continue
        visited.add(node)
        
        # 松弛操作
        for neighbor, weight in graph[node]:
            new_dist = dist[node] + weight
            if new_dist < dist[neighbor]:
                dist[neighbor] = new_dist
                heapq.heappush(pq, (new_dist, neighbor))
    
    return dist
```

**时间复杂度：** O((V + E) log V)  
**⚠️ 限制：** 不能处理负权边

**典型例题：**
- LeetCode 743. 网络延迟时间
- LeetCode 787. K站中转内最便宜的航班

## 4. Bellman-Ford 算法

```python
def bellman_ford(n, edges, start):
    """
    可以处理负权边，可以检测负环
    n: 节点数量 (0 to n-1)
    edges: [(u, v, weight), ...]
    start: 起点
    """
    dist = [float('inf')] * n
    dist[start] = 0
    
    # 松弛 n-1 次
    for _ in range(n - 1):
        for u, v, w in edges:
            if dist[u] != float('inf') and dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
    
    # 检测负环
    for u, v, w in edges:
        if dist[u] != float('inf') and dist[u] + w < dist[v]:
            return None  # 存在负环
    
    return dist
```

**时间复杂度：** O(V × E)

## 5. Floyd-Warshall 算法

```python
def floyd_warshall(n, edges):
    """
    所有节点对之间的最短路径
    n: 节点数量
    edges: [(u, v, weight), ...]
    返回: 二维数组 dist[i][j]
    """
    # 初始化
    dist = [[float('inf')] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    for u, v, w in edges:
        dist[u][v] = w
    
    # 动态规划
    for k in range(n):
        for i in range(n):
            for j in range(n):
                dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j])
    
    return dist
```

**时间复杂度：** O(V³)  
**典型例题：** LeetCode 1334. 阈值距离内邻居最少的城市

## 6. 最小生成树 - Kruskal

```python
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        root_x, root_y = self.find(x), self.find(y)
        if root_x == root_y:
            return False
        self.parent[root_x] = root_y
        return True

def kruskal(n, edges):
    """
    Kruskal算法 - 最小生成树
    n: 节点数量
    edges: [(u, v, weight), ...]
    """
    # 按权重排序
    edges.sort(key=lambda x: x[2])
    
    uf = UnionFind(n)
    mst_weight = 0
    mst_edges = []
    
    for u, v, w in edges:
        if uf.union(u, v):
            mst_weight += w
            mst_edges.append((u, v, w))
            if len(mst_edges) == n - 1:
                break
    
    return mst_weight, mst_edges
```

**时间复杂度：** O(E log E)

## 7. 二分图判定

```python
from collections import deque

def is_bipartite(graph):
    """
    判断图是否为二分图
    使用染色法：BFS
    """
    n = len(graph)
    color = [-1] * n
    
    def bfs(start):
        queue = deque([start])
        color[start] = 0
        
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if color[neighbor] == -1:
                    color[neighbor] = 1 - color[node]
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:
                    return False
        return True
    
    # 处理多个连通分量
    for i in range(n):
        if color[i] == -1:
            if not bfs(i):
                return False
    return True
```

**典型例题：** LeetCode 785. 判断二分图

## 8. 强连通分量 - Tarjan算法

```python
def tarjan_scc(graph):
    """
    Tarjan算法求强连通分量
    """
    n = len(graph)
    dfn = [-1] * n  # 访问时间戳
    low = [-1] * n  # 能到达的最小时间戳
    on_stack = [False] * n
    stack = []
    scc_list = []
    timestamp = [0]
    
    def dfs(node):
        dfn[node] = low[node] = timestamp[0]
        timestamp[0] += 1
        stack.append(node)
        on_stack[node] = True
        
        for neighbor in graph[node]:
            if dfn[neighbor] == -1:
                dfs(neighbor)
                low[node] = min(low[node], low[neighbor])
            elif on_stack[neighbor]:
                low[node] = min(low[node], dfn[neighbor])
        
        # 找到强连通分量的根
        if dfn[node] == low[node]:
            scc = []
            while True:
                v = stack.pop()
                on_stack[v] = False
                scc.append(v)
                if v == node:
                    break
            scc_list.append(scc)
    
    for i in range(n):
        if dfn[i] == -1:
            dfs(i)
    
    return scc_list
```

## 9. 网络流 - 最大流（Ford-Fulkerson）

```python
from collections import deque

def max_flow(graph, source, sink):
    """
    Edmonds-Karp算法（BFS实现的Ford-Fulkerson）
    graph: 邻接矩阵，graph[u][v]表示u->v的容量
    """
    n = len(graph)
    residual = [row[:] for row in graph]  # 残余图
    
    def bfs():
        """找增广路径"""
        visited = [False] * n
        parent = [-1] * n
        queue = deque([source])
        visited[source] = True
        
        while queue:
            u = queue.popleft()
            for v in range(n):
                if not visited[v] and residual[u][v] > 0:
                    visited[v] = True
                    parent[v] = u
                    queue.append(v)
                    if v == sink:
                        return parent
        return None
    
    max_flow_value = 0
    
    while True:
        parent = bfs()
        if not parent:
            break
        
        # 找最小残余容量
        path_flow = float('inf')
        v = sink
        while v != source:
            u = parent[v]
            path_flow = min(path_flow, residual[u][v])
            v = u
        
        # 更新残余图
        v = sink
        while v != source:
            u = parent[v]
            residual[u][v] -= path_flow
            residual[v][u] += path_flow
            v = u
        
        max_flow_value += path_flow
    
    return max_flow_value
```

## 图算法选择指南

| 问题类型 | 推荐算法 | 复杂度 |
|---------|---------|--------|
| 单源最短路（无负权） | Dijkstra | O(E log V) |
| 单源最短路（有负权） | Bellman-Ford | O(VE) |
| 所有点对最短路 | Floyd-Warshall | O(V³) |
| 最小生成树 | Kruskal/Prim | O(E log E) |
| 拓扑排序 | Kahn/DFS | O(V + E) |
| 强连通分量 | Tarjan/Kosaraju | O(V + E) |
| 二分图判定 | 染色法 | O(V + E) |
