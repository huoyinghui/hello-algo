## 一、哈希表基础

### 1.1 哈希表原理

**核心**: 通过哈希函数将键映射到数组下标,实现O(1)的查找、插入、删除

**特点**:
- ✅ 查找 O(1)
- ✅ 插入 O(1)
- ✅ 删除 O(1)
- ❌ 无序（Python 3.7+保证插入顺序）
- ❌ 空间换时间

### 1.2 哈希冲突解决

1. **链地址法**: 冲突的元素用链表连接
2. **开放寻址法**: 寻找下一个空位

## 二、Python字典dict

### 2.1 基本操作🐍

```python
# 创建字典
d = {}
d = dict()
d = {'a': 1, 'b': 2}
d = dict(a=1, b=2)

# 访问
value = d['a']        # KeyError if not exists
value = d.get('a')    # None if not exists
value = d.get('a', 0) # 默认值

# 插入/更新
d['c'] = 3
d.update({'d': 4, 'e': 5})

# 删除
del d['a']
value = d.pop('b')     # 删除并返回
d.popitem()            # 删除最后一项（3.7+）
d.clear()              # 清空

# 判断键是否存在
if 'a' in d:
    # ...

# 遍历
for key in d:           # 遍历键
    # ...
for value in d.values():  # 遍历值
    # ...
for key, value in d.items():  # 遍历键值对🔥
    # ...
```

### 2.3 Counter🐍

```python
from collections import Counter

# 计数器
nums = [1, 2, 2, 3, 3, 3]
counter = Counter(nums)  # Counter({3: 3, 2: 2, 1: 1})

# 最常见的N个元素
most_common = counter.most_common(2)  # [(3, 3), (2, 2)]

# 字符串计数
s = "hello"
counter = Counter(s)  # Counter({'l': 2, 'h': 1, 'e': 1, 'o': 1})

# 算术运算
c1 = Counter(a=3, b=1)
c2 = Counter(a=1, b=2)
c1 + c2  # Counter({'a': 4, 'b': 3})
c1 - c2  # Counter({'a': 2})
```

## 三、Python集合set

### 3.1 基本操作🐍

```python
# 创建集合
s = set()
s = {1, 2, 3}
s = set([1, 2, 2, 3])  # {1, 2, 3} 自动去重

# 添加
s.add(4)
s.update([5, 6])  # 批量添加

# 删除
s.remove(1)    # KeyError if not exists
s.discard(1)   # 不存在也不报错
s.pop()        # 随机删除一个
s.clear()      # 清空

# 判断存在
if 1 in s:
    # ...

# 遍历
for item in s:
    # ...
```

### 3.2 集合运算🐍

```python
s1 = {1, 2, 3}
s2 = {2, 3, 4}

# 并集
s1 | s2        # {1, 2, 3, 4}
s1.union(s2)

# 交集
s1 & s2        # {2, 3}
s1.intersection(s2)

# 差集
s1 - s2        # {1}
s1.difference(s2)

# 对称差集
s1 ^ s2        # {1, 4}
s1.symmetric_difference(s2)

# 子集/超集判断
{1, 2} <= {1, 2, 3}  # True (子集)
{1, 2, 3} >= {1, 2}  # True (超集)
```

## 四、经典题目

### 4.1 两数之和🔥

```python
def two_sum(nums, target):
    seen = {}  # 值 -> 索引
    for i, num in enumerate(nums):
        complement = target - num
        if complement in seen:
            return [seen[complement], i]
        seen[num] = i
    return []
```

### 4.2 字母异位词分组🔥

```python
from collections import defaultdict

def group_anagrams(strs):
    groups = defaultdict(list)
    
    for s in strs:
        # 方法1: 排序作为key
        key = ''.join(sorted(s))
        groups[key].append(s)
    
    return list(groups.values())

# 方法2: 字符计数作为key🐍
def group_anagrams_v2(strs):
    groups = defaultdict(list)
    
    for s in strs:
        count = [0] * 26
        for char in s:
            count[ord(char) - ord('a')] += 1
        groups[tuple(count)].append(s)  # list不能作key,用tuple
    
    return list(groups.values())
```

### 4.3 最长连续序列🔥

```python
def longest_consecutive(nums):
    num_set = set(nums)
    max_len = 0
    
    for num in num_set:
        # 只从序列起点开始🔥
        if num - 1 not in num_set:
            current_num = num
            current_len = 1
            
            while current_num + 1 in num_set:
                current_num += 1
                current_len += 1
            
            max_len = max(max_len, current_len)
    
    return max_len
```

### 4.4 有效的数独

```python
def is_valid_sudoku(board):
    rows = [set() for _ in range(9)]
    cols = [set() for _ in range(9)]
    boxes = [set() for _ in range(9)]
    
    for i in range(9):
        for j in range(9):
            num = board[i][j]
            if num == '.':
                continue
            
            box_index = (i // 3) * 3 + j // 3
            
            if (num in rows[i] or 
                num in cols[j] or 
                num in boxes[box_index]):
                return False
            
            rows[i].add(num)
            cols[j].add(num)
            boxes[box_index].add(num)
    
    return True
```

### 4.5 两个数组的交集

```python
# 交集I（去重）
def intersection(nums1, nums2):
    return list(set(nums1) & set(nums2))

# 交集II（保留重复）
from collections import Counter

def intersect(nums1, nums2):
    counter1 = Counter(nums1)
    counter2 = Counter(nums2)
    
    # 取两个Counter的最小值🐍
    intersection = counter1 & counter2
    return list(intersection.elements())
```

## 五、高级技巧

### 5.1 哈希表 + 前缀和🔥

#### 📌 和为K的子数组

```python
from collections import defaultdict

def subarray_sum(nums, k):
    count = 0
    prefix_sum = 0
    prefix_count = defaultdict(int)
    prefix_count[0] = 1  # 初始化🔥
    
    for num in nums:
        prefix_sum += num
        count += prefix_count[prefix_sum - k]
        prefix_count[prefix_sum] += 1
    
    return count
```

#### 📌 连续数组（0和1数量相等）

```python
def find_max_length(nums):
    # 把0看作-1,问题转化为和为0的最长子数组
    prefix_sum = 0
    sum_index = {0: -1}  # 前缀和 -> 最早出现的索引
    max_len = 0
    
    for i, num in enumerate(nums):
        prefix_sum += 1 if num == 1 else -1
        
        if prefix_sum in sum_index:
            max_len = max(max_len, i - sum_index[prefix_sum])
        else:
            sum_index[prefix_sum] = i
    
    return max_len
```

### 5.2 哈希表 + 滑动窗口🔥

#### 📌 无重复字符的最长子串

```python
def length_of_longest_substring(s):
    char_index = {}  # 字符 -> 最后出现的索引
    left = 0
    max_len = 0
    
    for right, char in enumerate(s):
        if char in char_index and char_index[char] >= left:
            left = char_index[char] + 1
        
        char_index[char] = right
        max_len = max(max_len, right - left + 1)
    
    return max_len
```

### 5.3 哈希表 + 双指针🔥

#### 📌 四数相加II

```python
from collections import defaultdict

def four_sum_count(nums1, nums2, nums3, nums4):
    # 分成两组,降低复杂度🔥
    sum_count = defaultdict(int)
    
    # 统计nums1和nums2的两数之和
    for a in nums1:
        for b in nums2:
            sum_count[a + b] += 1
    
    # 统计nums3和nums4中有多少对和为 -(c+d)
    count = 0
    for c in nums3:
        for d in nums4:
            count += sum_count[-(c + d)]
    
    return count
```
