## 一、栈 Stack

### 1.1 Python实现🐍

```python
# 方法1: 用list实现（最常用）🐍
stack = []
stack.append(1)      # 入栈 O(1)
stack.append(2)
top = stack[-1]      # 查看栈顶 O(1)
val = stack.pop()    # 出栈 O(1)
is_empty = len(stack) == 0

# 方法2: 用deque实现（更高效）
from collections import deque
stack = deque()
stack.append(1)
stack.pop()
```

### 1.2 栈的经典应用

#### 📌 有效的括号🔥

```python
def is_valid(s):
    stack = []
    mapping = {')': '(', '}': '{', ']': '['}
    
    for char in s:
        if char in mapping:  # 右括号
            if not stack or stack[-1] != mapping[char]:
                return False
            stack.pop()
        else:  # 左括号
            stack.append(char)
    
    return len(stack) == 0
```

#### 📌 最小栈

```python
class MinStack:
    def __init__(self):
        self.stack = []
        self.min_stack = []  # 维护最小值🔥
    
    def push(self, val):
        self.stack.append(val)
        if not self.min_stack or val <= self.min_stack[-1]:
            self.min_stack.append(val)
    
    def pop(self):
        val = self.stack.pop()
        if val == self.min_stack[-1]:
            self.min_stack.pop()
    
    def top(self):
        return self.stack[-1]
    
    def get_min(self):
        return self.min_stack[-1]
```

#### 📌 逆波兰表达式求值

```python
def eval_rpn(tokens):
    stack = []
    operators = {'+', '-', '*', '/'}
    
    for token in tokens:
        if token in operators:
            b = stack.pop()
            a = stack.pop()
            if token == '+':
                stack.append(a + b)
            elif token == '-':
                stack.append(a - b)
            elif token == '*':
                stack.append(a * b)
            else:
                stack.append(int(a / b))  # 向零截断🐍
        else:
            stack.append(int(token))
    
    return stack[0]
```

## 二、队列 Queue

### 2.1 Python实现🐍

```python
# 方法1: 用deque实现（推荐）🐍
from collections import deque

queue = deque()
queue.append(1)       # 入队 O(1)
queue.append(2)
val = queue.popleft() # 出队 O(1)
front = queue[0]      # 查看队首
is_empty = len(queue) == 0

# 方法2: 用list实现（不推荐，pop(0)是O(n)）
queue = []
queue.append(1)
queue.pop(0)  # ❌ O(n)操作
```

### 2.2 队列的应用

#### 📌 用栈实现队列

```python
class MyQueue:
    def __init__(self):
        self.in_stack = []   # 输入栈
        self.out_stack = []  # 输出栈
    
    def push(self, x):
        self.in_stack.append(x)
    
    def pop(self):
        self._move()
        return self.out_stack.pop()
    
    def peek(self):
        self._move()
        return self.out_stack[-1]
    
    def empty(self):
        return not self.in_stack and not self.out_stack
    
    def _move(self):
        # 只有out_stack为空时才移动🔥
        if not self.out_stack:
            while self.in_stack:
                self.out_stack.append(self.in_stack.pop())
```

#### 📌 用队列实现栈

```python
class MyStack:
    def __init__(self):
        self.queue = deque()
    
    def push(self, x):
        self.queue.append(x)
        # 把新元素前面的元素都移到后面🔥
        for _ in range(len(self.queue) - 1):
            self.queue.append(self.queue.popleft())
    
    def pop(self):
        return self.queue.popleft()
    
    def top(self):
        return self.queue[0]
    
    def empty(self):
        return len(self.queue) == 0
```

#### 📌 循环队列

```python
class MyCircularQueue:
    def __init__(self, k):
        self.queue = [0] * k
        self.head = 0
        self.tail = 0
        self.size = 0
        self.capacity = k
    
    def enqueue(self, value):
        if self.is_full():
            return False
        self.queue[self.tail] = value
        self.tail = (self.tail + 1) % self.capacity
        self.size += 1
        return True
    
    def dequeue(self):
        if self.is_empty():
            return False
        self.head = (self.head + 1) % self.capacity
        self.size -= 1
        return True
    
    def front(self):
        return -1 if self.is_empty() else self.queue[self.head]
    
    def rear(self):
        return -1 if self.is_empty() else self.queue[(self.tail - 1) % self.capacity]
    
    def is_empty(self):
        return self.size == 0
    
    def is_full(self):
        return self.size == self.capacity
```

## 三、单调栈

### 3.1 单调栈原理🔥

**核心思想**: 维护一个单调递增或单调递减的栈,用于快速找到"下一个更大/更小元素"

**应用场景**:
- 下一个更大元素
- 柱状图最大矩形
- 去重保留字典序最小

### 3.2 单调栈模板

```python
# 模板: 找下一个更大元素
def next_greater_element(nums):
    n = len(nums)
    result = [-1] * n
    stack = []  # 单调递减栈（存下标）
    
    for i in range(n):
        # 当前元素比栈顶大，找到了答案
        while stack and nums[stack[-1]] < nums[i]:
            result[stack.pop()] = nums[i]
        stack.append(i)
    
    return result
```

### 3.3 单调栈经典题

#### 📌 下一个更大元素 I

```python
def next_greater_element(nums1, nums2):
    # 先计算nums2中每个元素的下一个更大元素
    next_greater = {}
    stack = []
    
    for num in nums2:
        while stack and stack[-1] < num:
            next_greater[stack.pop()] = num
        stack.append(num)
    
    # 查询nums1的答案🐍
    return [next_greater.get(num, -1) for num in nums1]
```

#### 📌 每日温度🔥

```python
def daily_temperatures(temperatures):
    n = len(temperatures)
    result = [0] * n
    stack = []  # 单调递减栈
    
    for i in range(n):
        while stack and temperatures[stack[-1]] < temperatures[i]:
            prev_index = stack.pop()
            result[prev_index] = i - prev_index
        stack.append(i)
    
    return result
```

#### 📌 柱状图中最大的矩形🔥

```python
def largest_rectangle_area(heights):
    stack = []
    max_area = 0
    heights = [0] + heights + [0]  # 哨兵🐍
    
    for i, h in enumerate(heights):
        while stack and heights[stack[-1]] > h:
            height = heights[stack.pop()]
            width = i - stack[-1] - 1
            max_area = max(max_area, height * width)
        stack.append(i)
    
    return max_area
```

#### 📌 去除重复字母（字典序最小）🔥

```python
from collections import Counter

def remove_duplicate_letters(s):
    count = Counter(s)  # 统计每个字符剩余次数🐍
    in_stack = set()
    stack = []
    
    for char in s:
        count[char] -= 1
        
        if char in in_stack:
            continue
        
        # 如果当前字符更小，且栈顶字符后面还有，就弹出
        while stack and stack[-1] > char and count[stack[-1]] > 0:
            in_stack.remove(stack.pop())
        
        stack.append(char)
        in_stack.add(char)
    
    return ''.join(stack)
```

## 四、单调队列

### 4.1 单调队列原理🔥

**核心思想**: 维护一个单调队列,用于快速找到滑动窗口内的最大/最小值

**应用场景**:
- 滑动窗口最大值
- 滑动窗口中位数

### 4.2 单调队列模板

```python
from collections import deque

# 滑动窗口最大值模板
def max_sliding_window(nums, k):
    dq = deque()  # 存储下标,保持单调递减🐍
    result = []
    
    for i, num in enumerate(nums):
        # 移除窗口外的元素
        if dq and dq[0] < i - k + 1:
            dq.popleft()
        
        # 维护单调递减,移除比当前小的
        while dq and nums[dq[-1]] < num:
            dq.pop()
        
        dq.append(i)
        
        # 窗口形成后记录最大值
        if i >= k - 1:
            result.append(nums[dq[0]])
    
    return result
```
