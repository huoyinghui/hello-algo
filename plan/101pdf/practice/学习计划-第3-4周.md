## 🔥 核心模板

### 动态规划模板🔥
```python
# 1. 记忆化搜索🐍
from functools import lru_cache

@lru_cache(maxsize=None)
def dp(n):
    if n <= 1:
        return n
    return dp(n-1) + dp(n-2)

# 2. 自底向上DP
def dp_bottom_up(n):
    if n <= 1:
        return n
    dp = [0] * (n + 1)
    dp[1] = 1
    for i in range(2, n + 1):
        dp[i] = dp[i-1] + dp[i-2]
    return dp[n]

# 3. 空间优化
def dp_optimized(n):
    if n <= 1:
        return n
    prev, curr = 0, 1
    for _ in range(2, n + 1):
        prev, curr = curr, prev + curr
    return curr
```

### 回溯模板🔥
```python
def backtrack(路径, 选择列表):
    if 满足结束条件:
        result.append(路径.copy())
        return
    
    for 选择 in 选择列表:
        # 做选择
        路径.append(选择)
        
        # 递归
        backtrack(路径, 新的选择列表)
        
        # 撤销选择
        路径.pop()
```
